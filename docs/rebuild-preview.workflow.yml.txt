name: Rebuild preview (daily review-count refresh)
# INSTALL: copy this file to .github/workflows/rebuild-preview.yml (GitHub web UI: Add file > Create new file), or push with a token that has the `workflow` scope.
# Rebuilds the site so the live review counts (src/_data/reviews.js) stay current, then publishes to the
# gh-pages branch exactly like deploy-preview.sh (same PATH_PREFIX + CSS check).
# Optional secret: GOOGLE_PLACES_API_KEY (Settings > Secrets and variables > Actions). Never commit it.
on:
  schedule:
    - cron: "17 9 * * *"   # daily, 09:17 UTC (~4:17 AM Central)
  workflow_dispatch:
permissions:
  contents: write
concurrency:
  group: rebuild-preview
  cancel-in-progress: false
jobs:
  rebuild:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 20, cache: npm }
      - run: npm ci
      - name: Build with the GitHub Pages prefix
        run: |
          rm -rf _site
          PATH_PREFIX=/celebrations-preview/ npx @11ty/eleventy
          grep -q 'href="/celebrations-preview/assets/css/site.css' _site/index.html || { echo "ABORT: CSS path missing prefix"; exit 1; }
        env:
          GOOGLE_PLACES_API_KEY: ${{ secrets.GOOGLE_PLACES_API_KEY }}
      - name: Publish to gh-pages
        run: |
          rm -rf /tmp/gp && cp -r _site /tmp/gp && cd /tmp/gp && touch .nojekyll
          git init -q -b gh-pages && git add -A
          git -c user.name=github-actions -c user.email=github-actions@users.noreply.github.com commit -qm "deploy (scheduled rebuild)"
          git push -qf "https://x-access-token:${{ github.token }}@github.com/${{ github.repository }}.git" gh-pages
